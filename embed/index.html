  const pal = document.querySelector('obpal-remote')
  pal.setScene({ nodes: [
    { id: 'cube', name: 'Cube', kind: 'object' },
  ] })
  requestAnimationFrame(function loop(now) {
    for (const p of pal.participants) {
      const f = pal.frame(now, p.id)
      if (pal.holding(p.id) === 'cube')
        cube.position.x += f.pad1[0] * 0.01
    }
    requestAnimationFrame(loop)
  })
</script>
