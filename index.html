Build with it

Any web page can take a phone as its controller. This page does, with the same SDK.

import { Remote } from '@obpal/host'

const remote = await Remote.create({ appName: 'My scene' })
remote.mountPairing(document.querySelector('#pair'))

requestAnimationFrame(function frame(now) {
  const f = remote.consume(now)
  if (f.pose?.touching) follow(f.pose.p) // the hand, in metres
  requestAnimationFrame(frame)
})
